\section{Repeated Cross-Sectional Data}
\sectionframe{Repeated Cross-Sectional Data}
\begin{frame}{Repeated Cross-Sectional (RCS) Data }
\begin{block}{}
    Many organizations and government agencies use repeated surveys to track/model some metric of interest
\end{block}
\begin{block}{}
    For instance, the World Food Programme conducts surveys every day to monitor and model household food security in a given country
\end{block}


\begin{block}{}
A given household is only surveyed once, resulting in this data consisting of non-overlapping \textit{repeated cross-sections}
\end{block}

\end{frame}

\begin{frame}{We can design transformations of white noise that are based on the entire "history" of $\{Z_t\}:$ $X_t = Z_t + \phi Z_{t-1} + \phi^2Z_{t-2} + \phi^3Z_{t-3} + \cdots $}

$$ X_t-\phi X_{t-1} = Z_t$$
\end{block}

\begin{frame}{Introducing the Backwards Operator $B$}
\begin{block}{The backwards operator behaves as follows:}
\centering
$$ BX_t = X_{t-1}, \qquad B^2X_t = X_{t-2}, \qquad B^aX_t = X_{t-a} $$

We can use this operator to re-express the equations for AR(1), MA(1), and ARMA(1,1) process
\end{block}

\begin{block}{\centering AR(1) \hspace{1in} MA(1) \hspace{1in} ARMA(1,1)}
\centering
$$(1-\phi B)X_t = Z_t \hspace{.6in} X_t = (1+\theta B)Z_t \hspace{.6in} (1-\phi B)X_t = (1+\theta B)Z_t$$
\end{block}

\end{frame}

\begin{frame}{Existing methods for analysis}

\begin{block}{Aggregating output and explanatory variables then fit a time series with covariates}
\begin{itemize}
    \item This results in individual information loss
    \item Performs poorly at estimating covariate effects if daily distribution has low variance
\end{itemize}
\end{block}
\begin{block}{Implementing an Expectation-Maximization (EM) Algorithm (Modugno et al. 2015)}
\begin{itemize}
    \item Makes similar assumptions about model
    \item Leverages the covariance matrices of $\mathbf{u}$ and $\mathbf{y}$
    \item Serves as our starting point
\end{itemize}
    
\end{block}

\end{frame}

\begin{frame}{The sum of an ARMA process and white noise is still an ARMA process}
    \begin{block}{From Box and Jenkins (1976), we have the following theorem:}

\textit{If $\{u_t\}$ follows an ARMA($p_1,q_1$) model, $\{a_t\}$ follows an ARMA($p_2,q_2$) and the two series are independent, then their sum $w_t = u_t + a_t$ follows an ARMA($p,q$) model in which $p \leq p_1 + p_2$ and $q \leq \text{max}(p_1 + q_2,p_2 + q_1)$. }
    \end{block}
    \begin{block}{White noise is an ARMA(0,0) process. From this we can make two relevant conclusions:}
    
    \begin{enumerate}
        \item The sum of an AR(1) process and white noise is an ARMA(1,1) process
        \item The new ARMA process has the same AR parameter $\phi$, but a new MA parameter $\theta_b$ and innovations $\{b_t\}$ with variance $\sigma_b^2$
    \end{enumerate}
    \end{block}
\end{frame}

\begin{frame}{Asymptotics of MLE for ARMA-Noise Parameters}

\begin{block}{The MLE's for $\phi$ and $\theta_b$ have the asymptotic distribution:}
\centering
    $$ \sqrt{T}\left[\left(\begin{array}{c}
     \hat{\phi}  \\
      \hat{\theta}_b
\end{array}\right)-\left(\begin{array}{c}
     \phi  \\
     \theta_b
\end{array}\right)\right] \xrightarrow{d} N(\mathbf{0},\Sigma_{\phi,\theta_b})$$
\end{block}
\begin{block}{}
\centering

$$
\Sigma_{\phi,\theta_b} = \frac{1 + \phi\theta_b}{(\phi + \theta_b)^{2}} \left[ \begin{array}{cc}
    {(1 - \phi^2)(1 + \phi\theta_b)} & {-(1 - \theta_b^2)(1 - \phi^2)} \\
    {-(1 - \theta_b^2)(1 - \phi^2)} & {(1 - \theta_b^2)(1 + \phi\theta_b)}  
\end{array} \right] 
$$
\end{block}

    \begin{block}{With $\hat{\sigma}_\varepsilon^2$ defined as the pooled variance of residual deviations, we obtain our full asymptotic covariance matrix:}
        \centering
        $$\sqrt{T}\left[\left(\begin{array}{c}
     \hat{\phi}  \\
     \hat{\theta}_b \\
     \hat{\sigma}_b^2  \\
     \hat{\sigma}_\varepsilon^2
\end{array}\right) - \left(\begin{array}{c}
     {\phi}  \\
     {\theta}_b \\
     {\sigma}_b^2\\
     \sigma_\varepsilon^2
\end{array}\right)\right] \xrightarrow{d}N(\mathbf{0}, \Sigma_o), \qquad \Sigma_o = \left[ \begin{array}{ccc}
    \Sigma_{\phi,\theta_b} & \bm{0} & \mathbf{0} \\
    \mathbf{0} & {2\sigma_b^4} & 0 \\
   \mathbf{0}& 0 & \frac{2\sigma_\varepsilon^4}{n-1} 
\end{array} \right] $$
    \end{block}



\end{frame}

\begin{frame}{Asymptotics of MLE for ARMA Parameters}

\begin{block}{From Wong and Miller (1990), let $H$ be the system of equations with inputs $\tau_o = (\phi, \theta_b, \sigma_b^2, \sigma_\varepsilon^2)$ and outputs $\hat{\tau}_l = (\hat{\theta}, \hat{\sigma}_\eta^2)$ defined as its roots. By a combination of the Delta Method and the Implicit Function Theorem, such estimates have the asymptotic distribution:}
    \centering
    $$\sqrt{T}(\hat{\tau}_l - \tau_l) \xrightarrow{d} N(\mathbf{0},G\Sigma_oG'), \qquad G = -\left(\frac{\partial H}{\partial \bm{\tau}_l}\right)^{-1}
  \left(\frac{\partial H}{\partial \bm{\tau}_o}\right)$$
\end{block}
\begin{block}{Now that we have asymptotically unbiased estimators for all noise parameters, we forego extracting $\hat{\mathbf{u}}$ and estimate $\beta$ directly with the GLS estimator:}
\centering
$$ \hat{\beta}_{GLS} = (X_f'\hat{\Omega}^{-1}X_f)^{-1}X_f'\hat{\Omega}^{-1}\mathbf{y} $$
\end{block}
\end{frame}

\begin{frame}[t]{We now present and implement our new REMARCS algorithm}

\begin{block}{REgression Models for Autocorrelated Repeated Cross-Sections}
    1. Estimate ${\beta}$ with $\hat{\beta} = (X_f'X_f)^{-1}X_f'\mathbf{y}$
\end{block}
\begin{block}{}
    2. Estimate $X_t\uu + \varepsilon$ with $\mathbf{y} - X_f\hat{\beta}$ and fit ARMA-Noise model to obtain estimates $(\hat{\phi}, \hat{\theta}, \hat{\sigma_\eta^2}, \hat{\sigma_\varepsilon^2})$
\end{block}
\begin{block}{}
    3. Use parameter estimates to calculate $\hat{\Omega}$;
\end{block}
\begin{block}{}
    4. Refit fixed effect parameter estimates by calculating GLS estimate $\hat{\beta}_{GLS} = (X_f'\hat{\Omega}^{-1}X_f)X_f'\hat{\Omega}^{-1}\mathbf{y}$
\end{block}
\begin{block}{}
    5. Repeat Steps 2-4 until convergence

\section{Computational Methods}
\sectionframe{Computational Methods}
\begin{frame}{We can expedite the process by leveraging covariance structure}

\begin{block}{Inverting the estimate for $\Omega = \text{Var}(\mathbf{y})$ is the main bottleneck}

{Recall:}
\centering$$\Omega = X_t\Gamma X_t + \sigma_\varepsilon^2 I_N$$
    
\end{block}

\begin{block}{Define $\mathbf{p}$ and $\lambda$ such that:}
\centering
\[ \Gamma \mathbf{p} = \lambda \mathbf{p} \]
\end{block}

\begin{block}{$X_t\mathbf{p}$ is an eigenvector of $X_t\Gamma X_t'$ with corresponding eigenvalue $n\lambda$}
\centering
$$ (X_t \Gamma X_t')(X_t\mathbf{p}_i) = (X_t\Gamma)(X_t'X_t)\mathbf{p}_i = X_t \Gamma n {I}_T\mathbf{p}_i = nX_t\Gamma \mathbf{p}_i = n\lambda (X_t \mathbf{p}_i) $$


\end{block}
\end{frame}



\begin{frame}{Frame Title}
\begin{block}{Defining $P$ and $\Lambda$ such that $\Gamma = P\Lambda P'$, we obtain:}
    \centering
    $$ X_t \Gamma X_t' = \left[\begin{array}{cc}
   n^{-1/2} X_t P & Q  
\end{array}\right] \left[\begin{array}{cc}
    n^{1/2}\sigma_\eta^2\Lambda & \mathbf{0} \\
    \mathbf{0} & \mathbf{0}
\end{array}\right] \left[\begin{array}{cc}
    n^{-1/2}P'X_t'  \\
    Q' 
\end{array}\right] $$
\end{block}

\begin{block}{$Q$ is defined as the computational cheap matrix of eigenvectors with corresponding eigenvalue $\lambda = 0$}

\centering
$$ \Omega = \left[\begin{array}{cc}
   n^{-1/2} X_t P & Q 
\end{array}\right] \left[\begin{array}{cc}
    n^{1/2}\sigma_\eta^2\Lambda + \sigma_\varepsilon^2I_T & \mathbf{0} \\
    \mathbf{0} & \sigma_\varepsilon^2I_N
\end{array}\right] \left[\begin{array}{cc}
    n^{-1/2}P'X_t'  \\
    Q' 
\end{array}\right]   $$
\end{block}

\begin{block}{We then obtain a computationally cheap expression for $\Omega$:}
    \centering
    $$\Omega^{-1} =  
   n^{-1} X_t P (n^{1/2}\sigma_\eta^2\Lambda + \sigma_\varepsilon^2I_T)^{-1}P'X_t' + \sigma_\varepsilon^{-2}Q
   Q'$$
\end{block}
\end{frame}











\section{Generalizing to Varying Sample Size}
\sectionframe{Generalizing to Varying Sample Size}

\begin{frame}{Adapting ARMA-Noise Models for varying sample size}
\begin{block}{}
{So far we have imposed the condition that the daily sample size be constant}
\end{block}

\begin{block}{}
{This is unrealistic in real RCS survey contexts}
\end{block}

\begin{block}{}
    {To what extent can we generalize the class of sample size distributions that still permit ARMA-Noise models to be fit?}
\end{block}
\begin{block}{}{We now consider daily sample sizes $\{n_t\}$ to be the realizations of random variables $\{N_t\}$}

\end{block}
\end{frame}
\begin{frame}{So long as $\mathbb{E}(N_t^{-1})$ is defined and constant for all $t$, ARMA-Noise models can still be fit}
\vspace{-.1in}

\begin{block}{\centering$w_{ti} = u_t + \varepsilon_{ti}, \qquad \varepsilon_{ti} \sim \text{WN}(\sigma_\varepsilon^2),\qquad i = 1,...,n_t, \qquad E(N_t^{-1}) = c \in (0,1)$
}
\centering
\[ a_t = n_t^{-1}\sum_{i=1}^{n_t}\varepsilon_{ti}, \qquad a_t \sim \text{WN}(\mathbb{E}(N_t^{-1})\sigma_\varepsilon^2) \]
\end{block}

\begin{block}{Our estimate for $\sigma_\varepsilon^2$ goes unchanged, whereas the aggregate noise variance estimate becomes:}
    \centering
    $$ \hat{\sigma}_a^2 = \frac{\hat{\sigma}_\varepsilon^2}{T}\sum_{t=1}^T\frac{1}{N_t} $$
\end{block}

\begin{block}{}
    I.I.D. and strictly stationary sample sizes necessarily yield white noise aggregates, whereas weakly stationary sample sizes do not
\end{block}

\end{frame}
